// =====================================================================================
// EURO SIGNAL : DIAGNOSTIC DU SCREENER, ProScreener ProRealTime (18/11/2026)
// À lancer sur la même liste et avec la même unité de temps que euro-signal-screener.txt.
// Affiche toujours des lignes : les 50 sociétés de la liste qui passent le plus de règles.
// Une capture suffit pour voir ce qui bloque.
//
// Colonne « Contrôle » : somme des points des règles passées ; 511 = toutes passées, la société doit
// alors sortir dans le screener (au moins en règle assouplie). Points de chaque règle :
//   256 historique de 61 semaines          128 au moins 3 M EUR échangés par jour
//    64 pas de saut anormal sur un an       32 volatilité sous 80 %/an
//    16 hausse sur 12 mois de +200 % au plus  8 hors du tiers le plus faible sur 12 mois
//     4 tendance (moyenne 10 semaines au-dessus de la 40, cours au-dessus de la 40)
//     2 pente du canal de +14 % en 6 mois au moins   1 hausse sur 12 mois de +30 % au moins
// Exemple : 383 = 511 - 128, seule la liquidité manque.
// =====================================================================================
TIMEFRAME(weekly)
histoOK = close[60] > 0
r1 = close / MAX(close[1], 0.0001) - 1
rc = MIN(MAX(r1, -0.5), 1)
vol52 = STD[52](rc) * SQRT(52)
mm10 = Average[10](close)
mm40 = Average[40](close)
tendance = mm10 > mm40 AND close > mm40
mom12 = close[4] / MAX(close[52], 0.0001) - 1
lc = LOG(close)
penteLog = LinearRegressionSlope[26](lc)
liquidite = close * Average[13](volume) / 5
sautSem = r1 > 1.5 OR r1 < -0.65
sansSaut = Summation[53](sautSem) = 0
rang12 = 1 / (1 + EXP((16 - mom12 * 100) / 19))
TIMEFRAME(default)
controle = 256 * histoOK + 128 * (liquidite >= 3000000) + 64 * sansSaut + 32 * (vol52 < 0.8) + 16 * (mom12 <= 2) + 8 * (rang12 >= 0.3333) + 4 * tendance + 2 * (penteLog >= LOG(1.3) / 52) + (mom12 >= 0.3)
c12 = ROUND(mom12 * 100)
c6 = ROUND((EXP(penteLog * 26) - 1) * 100)
SCREENER[close > 0](controle AS "Contrôle", c12 AS "12 mois %", c6 AS "Pente 6 mois %")
